Поч нельзя просто заммапить общее адресное пространство
Во-первых если процессы не родственные то не выйдет хранить в общей памяти указатели(региноы будут в разных вирт адресах)
Даже если процессы родственные то плохо то, что кэш с которым мы в бюольшинстве работаем он синхронизируется иногда с память физической, чего нам может быть не надо вообще

А что до самого пайпа, то достаточно просто всё
![](Pasted%20image%2020250124231141.png)
Давайте воспользуемся интерфейсом VFS, скажем что вот этот циклический буффер это файл, определим для него read() write()


```c++
#include <stdio.h>
#include <unistd.h>
#include <string.h>

int main() {
    int pipe_fd[2]; // Массив для файловых дескрипторов: pipe_fd[0] для чтения, pipe_fd[1] для записи
    pid_t pid;
    char write_msg[] = "Hello from parent process!";
    char read_msg[100];
    
    // Создание канала
    if (pipe(pipe_fd) == -1) {
        perror("Pipe failed");
        return 1;
    }

    // Создание дочернего процесса
    pid = fork();
    if (pid < 0) {
        perror("Fork failed");
        return 1;
    }

    if (pid > 0) { 
        // Родительский процесс
        close(pipe_fd[0]); // Закрываем конец для чтения
        write(pipe_fd[1], write_msg, strlen(write_msg) + 1); // Записываем данные в канал
        close(pipe_fd[1]); // Закрываем конец для записи
    } else {
        // Дочерний процесс
        close(pipe_fd[1]); // Закрываем конец для записи
        read(pipe_fd[0], read_msg, sizeof(read_msg)); // Читаем данные из канала
        printf("Child process received: %s\n", read_msg);
        close(pipe_fd[0]); // Закрываем конец для чтения
    }

    return 0;
}
```

```c++
#include <stdio.h>
#include <stdlib.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>

int main() {
    const char *fifo_name = "/tmp/myfifo";
    char buffer[100];
    
    // Создаем именованный пайп
    if (mkfifo(fifo_name, 0666) == -1) {
        perror("mkfifo failed");
        return 1;
    }

    if (fork() == 0) {
        // Дочерний процесс: читает из пайпа
        int fd = open(fifo_name, O_RDONLY);
        read(fd, buffer, sizeof(buffer));
        printf("Child received: %s\n", buffer);
        close(fd);
    } else {
        // Родительский процесс: пишет в пайп
        int fd = open(fifo_name, O_WRONLY);
        write(fd, "Hello from parent process!", 27);
        close(fd);
    }

    // Удаляем именованный пайп
    unlink(fifo_name);

    return 0;
}
```
